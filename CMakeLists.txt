cmake_minimum_required(VERSION 3.16)
project(StudentGradeCalculator VERSION 1.0 LANGUAGES CXX)
add_executable(student_grade_calculator main.cpp Student.cpp)
target_compile_features(student_grade_calculator PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(student_grade_calculator PRIVATE /W4)
else()
    target_compile_options(student_grade_calculator PRIVATE -Wall -Wextra -pedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_FOUND)
        add_test(NAME grade_calculator_checks
            COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/check.py $<TARGET_FILE:student_grade_calculator>)
    endif()
endif()
